import { jsx as _jsx, jsxs as _jsxs } from "react/jsx-runtime";
import { HUDItem, ProgressBar, Tooltip } from './BaseComponents';
export const HUD = ({ score, time, wordsFound, totalWords, hintsAvailable, difficulty, onHintClick, }) => {
    const minutes = Math.floor(time / 60);
    const seconds = time % 60;
    const timeFormatted = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
    const percentProgress = Math.round((wordsFound / totalWords) * 100);
    return (_jsxs("div", { className: "space-y-4", children: [_jsxs("div", { className: "hud", children: [_jsx(HUDItem, { label: "Pontos", value: score }), _jsx(HUDItem, { label: "Tempo", value: timeFormatted }), _jsx(HUDItem, { label: "Palavras", value: `${wordsFound}/${totalWords}` }), _jsxs("div", { className: "hud-section", children: [_jsx(Tooltip, { text: `Dificuldade: ${difficulty}`, children: _jsx("span", { className: "hud-label", children: "Dificuldade" }) }), _jsx("span", { className: "badge badge-primary text-xs", children: difficulty })] })] }), _jsxs("div", { className: "card-lg", children: [_jsxs("div", { className: "mb-2 flex justify-between items-center", children: [_jsx("span", { className: "text-sm text-muted", children: "Progresso" }), _jsxs("span", { className: "text-sm font-bold text-secondary", children: [percentProgress, "%"] })] }), _jsx(ProgressBar, { current: wordsFound, total: totalWords, animated: true })] }), _jsxs("div", { className: "card-lg flex items-center justify-between", children: [_jsxs("div", { children: [_jsx("span", { className: "text-sm text-muted block", children: "Dicas dispon\u00EDveis" }), _jsx("span", { className: "text-2xl font-bold text-secondary", children: hintsAvailable })] }), onHintClick && (_jsx("button", { onClick: onHintClick, disabled: hintsAvailable === 0, className: `px-4 py-2 rounded-lg font-medium transition-all ${hintsAvailable > 0
                            ? 'bg-secondary text-dark hover:scale-105'
                            : 'bg-gray-600 text-gray-300 cursor-not-allowed opacity-50'}`, children: "\uD83D\uDCA1 Dica" }))] })] }));
};
