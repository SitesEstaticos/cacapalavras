import { jsx as _jsx, jsxs as _jsxs, Fragment as _Fragment } from "react/jsx-runtime";
import { Button } from './BaseComponents';
export const Modal = ({ isOpen, onClose, title, children, actions }) => {
    if (!isOpen)
        return null;
    return (_jsx("div", { className: "modal-overlay", onClick: onClose, children: _jsxs("div", { className: "modal", onClick: e => e.stopPropagation(), children: [_jsxs("div", { className: "flex justify-between items-center mb-4", children: [_jsx("h2", { className: "text-2xl font-bold text-secondary", children: title }), _jsx("button", { onClick: onClose, className: "text-2xl leading-none hover:text-secondary transition-colors", children: "\u2715" })] }), _jsx("div", { className: "mb-6 text-light", children: children }), _jsx("div", { className: "flex gap-3 justify-end", children: actions || (_jsx(Button, { variant: "secondary", onClick: onClose, children: "Fechar" })) })] }) }));
};
export const HintModal = ({ isOpen, onClose, onUseHint, onWatchAd, hintsRemaining, hint, }) => {
    return (_jsx(Modal, { isOpen: isOpen, onClose: onClose, title: "Dica", children: _jsxs("div", { className: "space-y-4", children: [_jsx("div", { className: "p-4 rounded-lg bg-secondary bg-opacity-20 border border-secondary", children: _jsx("p", { className: "text-lg font-medium", children: hint }) }), hintsRemaining > 0 ? (_jsxs(_Fragment, { children: [_jsxs("p", { className: "text-sm text-muted", children: ["Voc\u00EA tem ", _jsx("span", { className: "text-secondary font-bold", children: hintsRemaining }), " dica", hintsRemaining !== 1 ? 's' : '', " gratuita", hintsRemaining !== 1 ? 's' : '', " hoje."] }), _jsxs("div", { className: "flex gap-2", children: [_jsx(Button, { variant: "primary", onClick: onUseHint, className: "flex-1", children: "Usar Dica" }), _jsx(Button, { variant: "ghost", onClick: onClose, className: "flex-1", children: "Cancelar" })] })] })) : (_jsxs(_Fragment, { children: [_jsx("p", { className: "text-sm text-muted", children: "Suas dicas gratuitas de hoje acabaram." }), _jsx("p", { className: "text-sm text-secondary font-medium", children: "Assista a um an\u00FAncio para ganhar +1 dica!" }), _jsxs("div", { className: "flex gap-2", children: [_jsx(Button, { variant: "secondary", onClick: onWatchAd, className: "flex-1", children: "\uD83D\uDCFA Assistir An\u00FAncio" }), _jsx(Button, { variant: "ghost", onClick: onClose, className: "flex-1", children: "Cancelar" })] })] }))] }) }));
};
export const GameOverModal = ({ isOpen, onClose, onRestart, score, time, wordsFound, totalWords, isVictory, }) => {
    const minutes = Math.floor(time / 60);
    const seconds = time % 60;
    return (_jsx(Modal, { isOpen: isOpen, onClose: onClose, title: isVictory ? '🎉 Vitória!' : '⏸️ Jogo Pausado', children: _jsxs("div", { className: "space-y-6", children: [_jsxs("div", { className: "grid grid-cols-2 gap-4", children: [_jsxs("div", { className: "text-center p-4 rounded-lg bg-white bg-opacity-5", children: [_jsx("p", { className: "text-xs text-muted mb-1", children: "Pontos" }), _jsx("p", { className: "text-3xl font-bold text-secondary", children: score })] }), _jsxs("div", { className: "text-center p-4 rounded-lg bg-white bg-opacity-5", children: [_jsx("p", { className: "text-xs text-muted mb-1", children: "Tempo" }), _jsxs("p", { className: "text-3xl font-bold text-secondary", children: [minutes, ":", seconds.toString().padStart(2, '0')] })] }), _jsxs("div", { className: "text-center p-4 rounded-lg bg-white bg-opacity-5", children: [_jsx("p", { className: "text-xs text-muted mb-1", children: "Palavras" }), _jsxs("p", { className: "text-3xl font-bold text-secondary", children: [wordsFound, "/", totalWords] })] }), _jsxs("div", { className: "text-center p-4 rounded-lg bg-white bg-opacity-5", children: [_jsx("p", { className: "text-xs text-muted mb-1", children: "Acur\u00E1cia" }), _jsxs("p", { className: "text-3xl font-bold text-secondary", children: [Math.round((wordsFound / totalWords) * 100), "%"] })] })] }), _jsxs("div", { className: "flex gap-3", children: [_jsx(Button, { variant: "secondary", onClick: onRestart, className: "flex-1", children: "Novo Jogo" }), _jsx(Button, { variant: "ghost", onClick: onClose, className: "flex-1", children: "Menu" })] })] }) }));
};
