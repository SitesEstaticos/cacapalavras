import { jsxs as _jsxs, jsx as _jsx } from "react/jsx-runtime";
export const WordList = ({ words, foundWords, onWordClick }) => {
    const found = words.filter(w => foundWords.includes(w.id));
    const remaining = words.filter(w => !foundWords.includes(w.id));
    return (_jsxs("div", { className: "card-lg space-y-4", children: [_jsxs("h3", { className: "text-lg font-bold text-secondary", children: ["Palavras (", found.length, "/", words.length, ")"] }), found.length > 0 && (_jsxs("div", { children: [_jsx("h4", { className: "text-xs uppercase text-muted mb-2", children: "Encontradas" }), _jsx("div", { className: "word-list", children: found.map(word => (_jsx("div", { className: "word-item-found cursor-pointer text-center font-medium text-sm", onClick: () => onWordClick?.(word.id), children: word.text }, word.id))) })] })), remaining.length > 0 && (_jsxs("div", { children: [_jsx("h4", { className: "text-xs uppercase text-muted mb-2", children: found.length > 0 ? 'Ainda faltam' : 'Encontre estas palavras' }), _jsx("div", { className: "word-list", children: remaining.map(word => (_jsx("div", { className: "word-item cursor-pointer hover:bg-white hover:bg-opacity-10 text-center font-medium text-sm", onClick: () => onWordClick?.(word.id), children: word.text }, word.id))) })] })), found.length === words.length && words.length > 0 && (_jsx("div", { className: "mt-4 p-4 rounded-lg bg-success bg-opacity-20 border border-success", children: _jsx("p", { className: "text-success font-bold text-center", children: "\uD83C\uDF89 Parab\u00E9ns! Jogo completo!" }) }))] }));
};
